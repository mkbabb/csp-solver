#!/usr/bin/env bash
# run: bash edge-headers.sh > edge-headers.txt   (read-only GETs/HEADs against the live edge; never deploys)
# A3 / T9-W8 8.1 — the edge's cache policy for / and every asset the LIVE index.html references.
set -u
BASE=https://sudoku.babb.dev
TMP=$(mktemp -d)
curl -sS "$BASE/" -o "$TMP/index.html" >/dev/null
ASSETS=$(grep -oE '(src|href)="/[^"]+"' "$TMP/index.html" | sed 's/.*="//;s/"//' | sort -u)
printf '%s\n' "== $(date -u +%Y-%m-%dT%H:%M:%SZ) — two passes per URL (pass 2 shows edge warmth) =="
for PASS in 1 2; do
  printf '\n--- pass %s ---\n' "$PASS"
  for U in "/" $ASSETS; do
    H=$(curl -sI --compressed "$BASE$U")
    code=$(printf '%s' "$H" | head -1 | tr -d '\r')
    get () { printf '%s' "$H" | tr -d '\r' | awk -F': ' -v k="$1" 'tolower($1)==k{print $2}' | head -1; }
    printf '%s\t%s\tcc=%s\tcf=%s\tage=%s\tetag=%s\tce=%s\tcl=%s\tvary=%s\tct=%s\n' \
      "$U" "$code" "$(get cache-control)" "$(get cf-cache-status)" "$(get age)" "$(get etag)" \
      "$(get content-encoding)" "$(get content-length)" "$(get vary)" "$(get content-type)"
  done
done
rm -rf "$TMP"
