THE COMPARATOR, inlined so this record is self-contained (it is a one-off probe,
not an estate script — it owes no lane, and a lane it does not own is a lane that
rots). Run from web/frontend with the two captures beside it.

    // Round-trip proof: the gate's own per-project resolution vs `--restamp --dry`'s proposal.
    import { readFileSync } from "node:fs";
    const S = process.argv[2];
    const parse = (text, kind) => {
      const rows = new Map();
      for (const line of text.split("\n")) {
        const m = /^ {2}(\S.*?)\s{2,}listed\s+(\d+)\s+live\s+(\d+)\s+derived\s+(\d+)\s+floor\s+(\d+)(?:\s*→\s*(\d+))?/.exec(line);
        if (!m) continue;
        rows.set(m[1], {
          listed: +m[2], live: +m[3], derived: +m[4],
          floor: kind === "restamp" ? +(m[6] ?? m[5]) : +m[5],
        });
      }
      return rows;
    };
    const gate = parse(readFileSync(`${S}/gate-resolution.txt`, "utf8"), "gate");
    const restamp = parse(readFileSync(`${S}/restamp-resolution.txt`, "utf8"), "restamp");
    const names = [...new Set([...gate.keys(), ...restamp.keys()])].sort();
    let bad = 0;
    console.log("project                     listed  live  derived  gate floor  restamp floor  verdict");
    for (const n of names) {
      const a = gate.get(n), b = restamp.get(n);
      const same = a && b && a.listed === b.listed && a.live === b.live && a.derived === b.derived && a.floor === b.floor;
      if (!same) bad += 1;
      console.log(
        `${n.padEnd(26)}  ${String(a?.listed).padStart(6)}  ${String(a?.live).padStart(4)}  ` +
        `${String(a?.derived).padStart(7)}  ${String(a?.floor).padStart(10)}  ${String(b?.floor).padStart(13)}  ` +
        (same ? "IDENTICAL" : "*** DIVERGES ***"),
      );
    }
    console.log(`\n${names.length} projects, ${bad} divergence(s).`);
    process.exit(bad ? 1 : 0);
