// PRM: live, because the row reads the EMPTY voice's box at rest in the regime that ships (no
// motion runs on an empty voice; a reduced-motion emulation would read a regime the owner does not see).
import { expect, test, type Browser, type Page } from "@playwright/test";

/**
 * THE NOTE'S RESERVE, READ ON THE PAGE (T9-W7 §7, NOTE-LEDGER; LAWS P7 §F (iii)).
 *
 * A LOCAL INSTRUMENT, NOT A CI LANE (O-12: CI is browserless; ratified for this row at T9-W7 pass 9).
 * CI's only reserve arm is `lint:ink`; this file is run by hand against a served build.
 *
 * Below 1024 an EMPTY voice must still hold its line: the push reads line one's rect as its
 * `from`, and a 0×0 voice flew the record in from nowhere. `lint:ink` carries the source law
 * (`check-ink-pressure.mjs`, the note's reserve row). A source reader cannot see every way the
 * voice can lose its box, so this row reads the RECT: at 390×844, 844×390 and 360×800 on a
 * coarse pointer, in both themes, the empty voice is at least its own line box tall and at least
 * the block's reserve.
 *
 * Each plant is a stylesheet or CSSOM edit written the way a source edit would land: scoped to
 * the component's own `data-v` attribute and appended after the product's sheet, so it wins by
 * order at equal specificity. Every plant must turn the verdict RED (M3 only in a portrait cell: in
 * landscape the moved seat still holds, so there it must stay GREEN), and the page must read GREEN
 * again once the plant is taken out. `display: contents`, a `:first-child` position and a role
 * are the three no source law can key on. A 1px font on the block is the negative: the block's
 * line is a length, so the reserve holds and the row must stay GREEN.
 */

// The section's classic easy 9×9 (30 givens), read back through the aria-label corpus.
const GIVENS: Record<number, number> = {
  0: 5, 1: 3, 4: 7, 9: 6, 12: 1, 13: 9, 14: 5, 19: 9, 20: 8, 25: 6,
  27: 8, 31: 6, 35: 3, 36: 4, 39: 8, 41: 3, 44: 1, 45: 7, 49: 2, 53: 6,
  55: 6, 60: 2, 61: 8, 66: 4, 67: 1, 68: 9, 71: 5, 76: 8, 79: 7, 80: 9,
};
const cells = Array.from({ length: 81 }, (_, i) => (GIVENS[i] ?? 0).toString(36)).join("");
const PAYLOAD = btoa(`\u00013.${cells}`)
  .replace(/\+/g, "-")
  .replace(/\//g, "_")
  .replace(/=+$/, "");

const CELLS = [
  { width: 390, height: 844 },
  { width: 844, height: 390 },
  { width: 360, height: 800 },
] as const;

type Reading = {
  text: string;
  height: number;
  line: number;
  reserve: number;
  display: string;
};

/** The empty voice's box against its own line and the block's reserve. */
const read = (page: Page): Promise<Reading> =>
  page.evaluate(() => {
    const voice = document.querySelector<HTMLElement>(".board-margin .margin-note");
    const block = voice?.closest<HTMLElement>(".margin-note-block");
    if (!voice || !block) throw new Error("no .board-margin .margin-note");
    const cs = getComputedStyle(voice);
    return {
      text: (voice.textContent ?? "").trim(),
      height: voice.getBoundingClientRect().height,
      line: parseFloat(cs.lineHeight),
      reserve: parseFloat(getComputedStyle(block).minHeight) || 0,
      display: cs.display,
    };
  });
const holds = (r: Reading) =>
  r.text === "" && r.height >= r.line - 0.5 && r.height >= r.reserve - 0.5;
const settle = (page: Page) =>
  page.evaluate(
    () => new Promise((done) => requestAnimationFrame(() => requestAnimationFrame(done))),
  );

/** A rule written as the SFC would write it: the component's scope attribute on each compound. */
type Plant = {
  name: string;
  css?: (v: string) => string;
  cssom?: "seat-to-landscape";
  /** the plant moves nothing in a landscape cell (the seat is still there), so it must stay GREEN */
  portraitOnly?: boolean;
};
const PLANTS: Plant[] = [
  {
    name: "M1 the block's reserve at 1px inside the <1024 block",
    css: (v) => `@media (max-width: 1023.98px) { .margin-note-block[${v}] { min-height: 1px; } }`,
  },
  {
    name: "M3 the seat moved to the landscape block only",
    cssom: "seat-to-landscape",
    portraitOnly: true,
  },
  {
    name: "M4 the block's line at 1px inside the <1024 block",
    css: (v) => `@media (max-width: 1023.98px) { .margin-note-block[${v}] { line-height: 1px; } }`,
  },
  {
    name: "M4em a 1.3em reserve on a 1px block",
    css: (v) =>
      `@media (max-width: 1023.98px) { .margin-note-block[${v}] { min-height: 1.3em; font-size: 1px; } }`,
  },
  {
    name: "M5 display: contents on the voice",
    css: (v) => `.margin-note[${v}] { display: contents; }`,
  },
  {
    name: "M6 `> p:first-child`",
    css: (v) => `.margin-note-block[${v}] > p:first-child { min-height: 0; }`,
  },
  {
    name: "M7 the voice reached by its role",
    css: (v) => `[role="status"][${v}] { min-height: 0; }`,
  },
];
const NEGATIVE = (v: string) =>
  `@media (max-width: 1023.98px) { .margin-note-block[${v}] { font-size: 1px; } }`;

/** The voice's scope attribute (`data-v-…`), read off the element the product rendered. */
const scopeOf = (page: Page) =>
  page.evaluate(() => {
    const voice = document.querySelector(".board-margin .margin-note");
    const attr = [...(voice?.attributes ?? [])].find((a) => a.name.startsWith("data-v-"));
    if (!attr) throw new Error("the voice carries no scope attribute");
    return attr.name;
  });

/** M3 through the CSSOM: the seat rule leaves the <1024 block and is re-inserted inside the
 *  landscape one. Returns an undo. */
async function seatToLandscape(page: Page, scope: string) {
  const moved = await page.evaluate((scope) => {
    const isSeat = (r: CSSRule) =>
      r instanceof CSSStyleRule &&
      r.selectorText === `.margin-note[${scope}]` &&
      r.style.getPropertyValue("min-height") === "inherit";
    for (const sheet of [...document.styleSheets]) {
      const rules = [...sheet.cssRules];
      const media = rules.find(
        (r): r is CSSMediaRule =>
          r instanceof CSSMediaRule &&
          /max-width: 1023\.98px/.test(r.conditionText) &&
          !/orientation/.test(r.conditionText) &&
          [...r.cssRules].some(isSeat),
      );
      const land = rules.find(
        (r): r is CSSMediaRule =>
          r instanceof CSSMediaRule &&
          /max-width: 1023\.98px/.test(r.conditionText) &&
          /orientation: landscape/.test(r.conditionText),
      );
      if (!media || !land) continue;
      const at = [...media.cssRules].findIndex(isSeat);
      const text = media.cssRules[at].cssText;
      media.deleteRule(at);
      land.insertRule(text, land.cssRules.length);
      (window as unknown as { __undoSeat: () => void }).__undoSeat = () => {
        land.deleteRule(land.cssRules.length - 1);
        media.insertRule(text, at);
      };
      return true;
    }
    return false;
  }, scope);
  if (!moved) throw new Error("M3: no seat rule found under the <1024 block");
  return () =>
    page.evaluate(() => (window as unknown as { __undoSeat: () => void }).__undoSeat());
}

async function openPhone(browser: Browser, browserName: string, cell: (typeof CELLS)[number], scheme: "light" | "dark") {
  const ctx = await browser.newContext({
    viewport: cell,
    deviceScaleFactor: 2,
    hasTouch: true,
    isMobile: browserName === "chromium",
    colorScheme: scheme,
  });
  const page = await ctx.newPage();
  await page.goto(`/?board=${PAYLOAD}`);
  // ATTACHED, not visible: the empty voice is 0 px wide by design (`width: fit-content`), so a
  // visibility wait never returns, and its height is the reading this row exists for.
  await page.waitForSelector(".board-margin .margin-note", { state: "attached", timeout: 30_000 });
  // the payload was dealt, not a board of the app's own (the aria-label corpus)
  await expect
    .poll(() =>
      page.evaluate(
        () =>
          [...document.querySelectorAll(".game-cell input")].filter((i) =>
            /given clue/.test(i.getAttribute("aria-label") ?? ""),
          ).length,
      ),
    )
    .toBe(Object.keys(GIVENS).length);
  const regime = await page.evaluate(() => ({
    coarse: matchMedia("(pointer: coarse)").matches,
    dark: matchMedia("(prefers-color-scheme: dark)").matches,
  }));
  expect(regime, "the coarse regime is witnessed, not assumed").toEqual({
    coarse: true,
    dark: scheme === "dark",
  });
  await settle(page);
  return { ctx, page };
}

for (const cell of CELLS) {
  test(`the empty voice holds its line at ${cell.width}×${cell.height} coarse, both themes, every plant red`, async ({
    browser,
    browserName,
  }) => {
    for (const scheme of ["light", "dark"] as const) {
      const { ctx, page } = await openPhone(browser, browserName, cell, scheme);
      const clean = await read(page);
      console.log(
        `NOTE-RESERVE|${browserName}|${cell.width}x${cell.height}|${scheme}|clean|${JSON.stringify(clean)}`,
      );
      expect(clean.text, `${scheme}: the voice is empty at the deal`).toBe("");
      // soft: a red clean read still lets the other theme run and print (the control's dark arm
      // is a reading, not a casualty of the light expect); its plants are not read on a red page
      expect.soft(holds(clean), `${scheme}: ${JSON.stringify(clean)}`).toBe(true);
      if (!holds(clean)) {
        await ctx.close();
        continue;
      }
      const scope = await scopeOf(page);

      for (const plant of PLANTS) {
        let undo: () => Promise<unknown>;
        if (plant.cssom) undo = await seatToLandscape(page, scope);
        else {
          const tag = await page.addStyleTag({ content: plant.css!(scope) });
          undo = () => tag.evaluate((el) => (el as Element).remove());
        }
        await settle(page);
        const planted = await read(page);
        await undo();
        await settle(page);
        const back = await read(page);
        const bites = !plant.portraitOnly || cell.height > cell.width;
        // the reading, printed beside the verdict (a row's transcript is its numbers)
        console.log(
          `NOTE-RESERVE|${browserName}|${cell.width}x${cell.height}|${scheme}|${plant.name}|${JSON.stringify(planted)}`,
        );
        expect(
          holds(planted),
          `${scheme} · ${plant.name} must ${bites ? "RED" : "stay GREEN here"}: ${JSON.stringify(planted)}`,
        ).toBe(!bites);
        expect(holds(back), `${scheme} · ${plant.name} taken out: ${JSON.stringify(back)}`).toBe(true);
      }

      const tag = await page.addStyleTag({ content: NEGATIVE(scope) });
      await settle(page);
      const negative = await read(page);
      await tag.evaluate((el) => (el as Element).remove());
      expect(holds(negative), `${scheme} · a 1px font on the block must stay GREEN: ${JSON.stringify(negative)}`).toBe(true);
      await ctx.close();
    }
  });
}

/**
 * REST = SPOKEN (pass 9, row 1). The reserve exists so that NOTHING under the voice moves when it
 * speaks: the strip (`.board-margin`) and the board's top read the same at rest and after the
 * first sentence. The empty-voice row above cannot see a rule that grows the line only once the
 * voice holds ink, so this row reads both instants. Its in-run negative is the double evasion
 * (an element-keyed line that grows on speech only); it must RED here, and the clean page must
 * read GREEN again on a fresh context. Three phone cells coarse and the 1280 desk fine, both themes.
 */
const SPOKEN_CELLS = [
  { width: 390, height: 844, coarse: true, dpr: 2 },
  { width: 844, height: 390, coarse: true, dpr: 2 },
  { width: 360, height: 800, coarse: true, dpr: 2 },
  { width: 1280, height: 800, coarse: false, dpr: 1 },
] as const;
type Spoken = { text: string; strip: number; boardTop: number; docH: number };
const readSpoken = (page: Page): Promise<Spoken> =>
  page.evaluate(() => {
    const voice = document.querySelector<HTMLElement>(".board-margin .margin-note");
    const strip = document.querySelector<HTMLElement>(".board-margin");
    const cell = document.querySelector<HTMLElement>(".game-cell");
    if (!voice || !strip || !cell) throw new Error("no voice, strip or board");
    return {
      text: (voice.textContent ?? "").trim(),
      strip: strip.getBoundingClientRect().height,
      boardTop: cell.getBoundingClientRect().top,
      docH: document.documentElement.scrollHeight,
    };
  });
/** Nothing under the voice moves: the strip and the board's top within half a pixel. */
const still = (rest: Spoken, spoke: Spoken) =>
  rest.text === "" &&
  spoke.text !== "" &&
  Math.abs(spoke.strip - rest.strip) <= 0.5 &&
  Math.abs(spoke.boardTop - rest.boardTop) <= 0.5;
/** The voice's first sentence: a hint on the first empty cell (the product's `h`). */
async function speak(page: Page) {
  await page.evaluate(() =>
    [...document.querySelectorAll<HTMLInputElement>(".game-cell input")]
      .find((i) => !i.value)
      ?.focus(),
  );
  await page.keyboard.press("h");
  await expect
    .poll(() => readSpoken(page).then((r) => r.text), { timeout: 10_000 })
    .not.toBe("");
  // the sentence's draw-in and the strip's settle; then the board is read at rest again
  await page.waitForTimeout(1600);
  await page.evaluate(() => (document.activeElement as HTMLElement | null)?.blur());
  await settle(page);
}
// The negative's rule, assembled at run time (a product string is a Tailwind source, T9-R6).
const DOUBLE = (v: string) =>
  `.margin-note-block[${v}] > p:first-child:${"has"}(.margin-note-ink) { line-height: 2.2; }`;
// The 0.5 px bound pinned from both sides (LAWS P8 §F): speech grows the strip by 0.4 px (inside the
// bound: still) and by 0.6 px (outside: moved). Light only; the bound is not a theme variable.
const EDGE = (px: number) => (v: string) =>
  `.margin-note-block[${v}] > p:first-child:${"has"}(.margin-note-ink) { padding-bottom: ${px}px; }`;
const PAD = (sel: string, decl: string) => (v: string) => `${sel.replace(/\$V/g, `[${v}]`)} { ${decl} }`;
const ARMS = {
  clean: { css: null, still: true, light: false },
  double: { css: DOUBLE, still: false, light: true },
  C1pad: { css: PAD(".margin-note-ink$V", "padding-block: 6px;"), still: false, light: true },
  C2inkfs: { css: PAD(".margin-note-ink$V", "font-size: 30px;"), still: false, light: true },
  C3voicepad: { css: PAD(".margin-note$V:" + "has(.margin-note-ink)", "padding-top: 8px;"), still: false, light: true },
  C6inklh: { css: PAD(".margin-note-ink$V", "line-height: 2.2;"), still: false, light: true },
  C7trans: { css: PAD(".note-swap-enter-active$V", "padding-bottom: 12px;"), still: false, light: true },
} as const;

async function openCell(
  browser: Browser,
  browserName: string,
  cell: (typeof SPOKEN_CELLS)[number],
  scheme: "light" | "dark",
) {
  const ctx = await browser.newContext({
    viewport: { width: cell.width, height: cell.height },
    deviceScaleFactor: cell.dpr,
    hasTouch: cell.coarse,
    isMobile: cell.coarse && browserName === "chromium",
    colorScheme: scheme,
  });
  const page = await ctx.newPage();
  await page.goto(`/?board=${PAYLOAD}`);
  await page.waitForSelector(".board-margin .margin-note", { state: "attached", timeout: 30_000 });
  await expect
    .poll(() =>
      page.evaluate(
        () =>
          [...document.querySelectorAll(".game-cell input")].filter((i) =>
            /given clue/.test(i.getAttribute("aria-label") ?? ""),
          ).length,
      ),
    )
    .toBe(Object.keys(GIVENS).length);
  const regime = await page.evaluate(() => ({
    coarse: matchMedia("(pointer: coarse)").matches,
    dark: matchMedia("(prefers-color-scheme: dark)").matches,
  }));
  expect(regime, "the regime is witnessed, not assumed").toEqual({
    coarse: cell.coarse,
    dark: scheme === "dark",
  });
  await page.waitForTimeout(1200);
  await settle(page);
  return { ctx, page };
}

for (const cell of SPOKEN_CELLS) {
  test(`rest = spoken at ${cell.width}×${cell.height} ${cell.coarse ? "coarse" : "fine"}, both themes, the double evasion red`, async ({
    browser,
    browserName,
  }) => {
    for (const scheme of ["light"] as const) {
      for (const [arm, spec] of Object.entries(ARMS)) {
        if (spec.light && scheme === "dark") continue;
        const { ctx, page } = await openCell(browser, browserName, cell, scheme);
        if (spec.css) await page.addStyleTag({ content: spec.css(await scopeOf(page)) });
        await settle(page);
        const rest = await readSpoken(page);
        await speak(page);
        const spoke = await readSpoken(page);
        const dStrip = spoke.strip - rest.strip;
        const dBoard = spoke.boardTop - rest.boardTop;
        console.log(
          `NOTE-SPOKEN|${browserName}|${cell.width}x${cell.height}|${scheme}|${arm}|strip ${rest.strip.toFixed(2)}->${spoke.strip.toFixed(2)} (${dStrip.toFixed(2)})|boardTop ${rest.boardTop.toFixed(2)}->${spoke.boardTop.toFixed(2)} (${dBoard.toFixed(2)})|docH ${rest.docH}->${spoke.docH}|${JSON.stringify(spoke.text)}`,
        );
        // the clean page is still; the double evasion must move the strip (its in-run negative)
        expect
          .soft(still(rest, spoke), `${scheme} · ${arm}: ${JSON.stringify({ rest, spoke })}`)
          .toBe(spec.still);
        await ctx.close();
      }
    }
  });
}

// CRITIC: the whole window, not two instants. Samples the strip every animation frame from the keypress for 2.5 s.
for (const arm of ["clean", "C7trans", "double"] as const) {
  test(`transient window ${arm} at 390x844`, async ({ browser, browserName }) => {
    const cell = SPOKEN_CELLS[0];
    const { ctx, page } = await openCell(browser, browserName, cell, "light");
    const spec = ARMS[arm];
    if (spec.css) await page.addStyleTag({ content: spec.css(await scopeOf(page)) });
    await settle(page);
    const rest = await readSpoken(page);
    await page.evaluate(() => {
      const w = window as unknown as { __h: number[] };
      w.__h = [];
      const strip = document.querySelector<HTMLElement>(".board-margin")!;
      const t0 = performance.now();
      const tick = () => {
        w.__h.push(strip.getBoundingClientRect().height);
        if (performance.now() - t0 < 2500) requestAnimationFrame(tick);
      };
      requestAnimationFrame(tick);
      [...document.querySelectorAll<HTMLInputElement>(".game-cell input")].find((i) => !i.value)?.focus();
    });
    await page.keyboard.press("h");
    await page.waitForTimeout(2800);
    const h = await page.evaluate(() => (window as unknown as { __h: number[] }).__h);
    const end = await readSpoken(page);
    const maxD = Math.max(...h.map((x) => Math.abs(x - rest.strip)));
    const moved = h.filter((x) => Math.abs(x - rest.strip) > 0.5).length;
    console.log(`NOTE-WINDOW|${browserName}|${arm}|rest ${rest.strip.toFixed(2)}|end ${end.strip.toFixed(2)}|frames ${h.length}|max|Δ| ${maxD.toFixed(2)}|frames moved>0.5 ${moved}|text ${JSON.stringify(end.text)}`);
    await ctx.close();
  });
}
