// CTRL-RULE pass 10 · T9-B13(b) struck to numbers (scratch probe, never banked).
// usage: node b13b-numbers.mjs <engine chromium|webkit> <arm-name> <url> [<arm-name> <url> ...]
// One browser per run, a new context per cell, every page/context/browser closed in finally.
// Cell: 390x844 coarse DPR 2, PRM, the default EASY deal, drawer open and settled, card at scroll END.
import { createRequire } from "node:module";
const require = createRequire("<repo>/web/frontend/package.json");
const pw = require("@playwright/test");
const [engine, ...rest] = process.argv.slice(2);
const arms = [];
for (let i = 0; i < rest.length; i += 2) arms.push([rest[i], rest[i + 1]]);
const browser = await pw[engine].launch({ headless: true });
let exit = 0;
try {
  for (const [arm, url] of arms)
    for (const theme of ["light", "dark"]) {
      const ctx = await browser.newContext({ viewport: { width: 390, height: 844 }, deviceScaleFactor: 2, isMobile: true, hasTouch: true, colorScheme: theme, reducedMotion: "reduce" });
      try {
        await ctx.addInitScript((t) => { try { localStorage.setItem("sudoku-color-scheme", t); } catch {} }, theme);
        const page = await ctx.newPage();
        await page.goto(url + "/?size=3&difficulty=EASY");
        await page.waitForSelector("svg.handwritten-logo", { timeout: 20000 });
        await page.waitForFunction(() => document.querySelectorAll(".sudoku-cell .glyph-svg").length > 0, null, { timeout: 20000 });
        if (await page.evaluate(() => document.documentElement.classList.contains("drawer-closed"))) await page.locator(".drawer-tab").tap();
        let last = NaN;
        for (let i = 0; i < 60; i++) {
          const t = await page.evaluate(() => document.querySelector(".drawer-case")?.getBoundingClientRect().top ?? NaN);
          if (Math.abs(t - last) < 0.01) break;
          last = t;
          await page.waitForTimeout(100);
        }
        const r = await page.evaluate(() => {
          const card = document.querySelector(".controls-card");
          card.scrollTop = card.scrollHeight;
          const cb = card.getBoundingClientRect();
          const vis = Math.max(0, Math.min(cb.bottom, innerHeight) - Math.max(cb.top, 0));
          const q = (s) => [...document.querySelectorAll(s)].filter((e) => e.getClientRects().length);
          const rules = q(".controls-card svg.ruled-line").map((e) => e.getBoundingClientRect());
          const foot = document.querySelector("#card-foot svg.ruled-line")?.getBoundingClientRect() ?? null;
          const inCardLast = rules.filter((b) => b.bottom <= cb.bottom + 0.5).sort((a, b) => a.bottom - b.bottom).pop() ?? null;
          const chip = q(".controls-card button, .controls-card [role=button]").find((e) => /\w/.test(e.textContent ?? ""));
          const word = (t) => q(".controls-card *").find((e) => e.children.length === 0 && (e.textContent ?? "").trim().toLowerCase() === t);
          const [ctr, mk] = [word("center"), word("marks")];
          const names = q(".rp-name");
          const deal = document.querySelector(".deal-row");
          const dealKids = deal ? [...deal.children].map((c) => c.getBoundingClientRect()) : [];
          const fieldL = deal ? deal.getBoundingClientRect().left : null;
          return {
            cardVisible: +vis.toFixed(1),
            ruledLines: rules.length,
            footRule: !!foot,
            blankBand: foot && inCardLast ? +(foot.top - inCardLast.bottom).toFixed(1) : null,
            names: names.length,
            nameInk: names[0] ? getComputedStyle(names[0]).color : null,
            tapes: q(".controls-card .washi-tag, .controls-card .zone-tape, .controls-card [class*='tape']").length,
            chipFont: chip ? getComputedStyle(chip).fontFamily.split(",")[0] : null,
            chipTag: chip ? `${chip.tagName}.${[...chip.classList].join(".")}` : null,
            ctrlWord: q(".controls-card .ctrl-word").length,
            ctrlBtn: q(".controls-card .ctrl-btn").length,
            centerWraps: ctr && mk ? +(ctr.getBoundingClientRect().top - mk.getBoundingClientRect().top).toFixed(1) : null,
            dealLeftInset: dealKids.length ? +(Math.min(...dealKids.map((b) => b.left)) - fieldL).toFixed(1) : null,
            dealRightInset: dealKids.length ? +(deal.getBoundingClientRect().right - Math.max(...dealKids.map((b) => b.right))).toFixed(1) : null,
          };
        });
        const js = [...(await page.content()).matchAll(/index-[A-Za-z0-9_-]+\.js/g)][0]?.[0];
        console.log(`[B13b] ${engine} ${arm} ${theme} ${js} ${JSON.stringify(r)}`);
      } catch (e) {
        exit = 1;
        console.log(`[B13b] ${engine} ${arm} ${theme} RED-THROW ${e}`);
      } finally {
        await ctx.close();
      }
    }
} finally {
  await browser.close();
}
process.exit(exit);
