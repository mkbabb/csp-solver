const CRIT_READ = async (page: Page) => {
  const rule = page.locator("#card-foot svg.ruled-line");
  const box = await rule.boundingBox();
  if (!box) return null;
  const clip = { x: box.x, y: box.y - 3, width: box.width, height: box.height + 6 };
  const on = (await page.screenshot({ clip })).toString("base64");
  const tag = await page.addStyleTag({ content: "#card-foot svg.ruled-line { visibility: hidden !important; }" });
  const off = (await page.screenshot({ clip })).toString("base64");
  await tag.evaluate((n) => (n as Element).remove());
  return page.evaluate(async ({ on, off, ch }) => {
    const px = async (b: string) => { const bmp = await createImageBitmap(await (await fetch(`data:image/png;base64,${b}`)).blob()); const c = new OffscreenCanvas(bmp.width, bmp.height); const g = c.getContext("2d")!; g.drawImage(bmp, 0, 0); return g.getImageData(0, 0, bmp.width, bmp.height); };
    const [a, b] = [await px(on), await px(off)];
    const { width: w, height: h } = a; const dpr = h / ch;
    const mv = (o: number) => Math.max(Math.abs(a.data[o] - b.data[o]), Math.abs(a.data[o + 1] - b.data[o + 1]), Math.abs(a.data[o + 2] - b.data[o + 2]));
    const all: number[] = []; for (let o = 0; o < a.data.length; o += 4) { const d = mv(o); if (d > 8) all.push(d); }
    all.sort((m, n) => m - n); const p98 = all.length ? all[Math.floor(0.98 * (all.length - 1))] : 1;
    const cols: number[] = [];
    for (let x = Math.floor(w * 0.05); x < Math.ceil(w * 0.95); x++) { let s = 0; for (let y = 0; y < h; y++) { const d = mv((y * w + x) * 4); if (d > 8) s += Math.min(1, d / p98); } cols.push(s / dpr); }
    const mid = cols.slice(Math.floor(cols.length * 0.2), Math.ceil(cols.length * 0.8)).sort((m, n) => m - n);
    return { median: +mid[mid.length >> 1].toFixed(3), p10: +mid[Math.floor(0.1 * (mid.length - 1))].toFixed(3), n: mid.length, dpr };
  }, { on, off, ch: clip.height });
};
