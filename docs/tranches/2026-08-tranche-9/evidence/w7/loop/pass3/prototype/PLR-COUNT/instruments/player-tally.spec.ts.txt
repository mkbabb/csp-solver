/**
 * THE TALLY (T9-W7 PLR-COUNT) — the count a reader takes without reading, as rows.
 *
 * One counting base: the strokes, the glyph, the accessible name, the register's rows and its
 * foot all count everyone at this board. One threshold: five. What this file holds is the
 * behaviour a screenshot cannot — the name against the glyph (2.5.3), the width table, the
 * departure that re-draws nothing, the depart-and-arrive that draws exactly one stroke, the
 * keys, the short phone that names two people, and the deck that has no mark at all.
 *
 * PRM: live, because every row here asserts a RESTING pose or a same-frame swap — the draw-in
 * is the one tween on this surface and no assertion reads it mid-flight; the reduced-motion
 * arm of the draw-in is asserted by `emulateMedia` below rather than inferred.
 *
 * LOCAL (O-12): CI is browserless; this is an instrument, run against a dev server by hand.
 */
import { test, expect, type Page } from "@playwright/test";

const mark = (page: Page) => page.locator("[data-player-mark]:visible").first();

/** The sheet fades and scales over 150ms; every box read here is of the SETTLED pose. */
const settle = (page: Page) => page.waitForTimeout(260);

async function settled(page: Page) {
  await page.waitForSelector("svg.handwritten-logo", { timeout: 60000 });
  await expect
    .poll(() => page.locator(".sudoku-cell .glyph-svg").count(), { timeout: 60000 })
    .toBeGreaterThan(0);
}

let cursor = 0;
/** k people arrive on the local arm. The page answers each ack and inks them from its own
 *  walk, so the indices are the room's real ones. */
async function peers(page: Page, room: string, k: number) {
  if (k <= 0) return;
  const from = cursor;
  cursor += k;
  await page.evaluate(
    ({ room, k, from }) => {
      const w = window as unknown as { __ch?: BroadcastChannel };
      w.__ch ??= new BroadcastChannel(`board:${room}`);
      for (let i = 0; i < k; i++)
        w.__ch.postMessage({ kind: "hi", data: {}, from: `tally-${from + i}` });
    },
    { room, k, from },
  );
  await page.waitForTimeout(700);
}

const head = (page: Page) =>
  page.evaluate(() => {
    const m = [...document.querySelectorAll("[data-player-mark]")].find(
      (e) => e.getBoundingClientRect().width > 0,
    ) as HTMLElement | undefined;
    const pose = m?.querySelector(".pt-pose");
    const b = m?.getBoundingClientRect();
    return {
      label: m?.getAttribute("aria-label") ?? null,
      width: b ? +b.width.toFixed(2) : null,
      height: b ? +b.height.toFixed(2) : null,
      strokes: pose ? pose.querySelectorAll("path").length : 0,
      written: m?.querySelector(".pt-count")?.textContent ?? null,
      writtenPx: m?.querySelector(".pt-count")
        ? getComputedStyle(m.querySelector(".pt-count")!).fontSize
        : null,
      ds: pose ? [...pose.querySelectorAll("path")].map((p) => p.getAttribute("d")) : [],
      offs: pose
        ? [...pose.querySelectorAll("path")].map((p) => p.getAttribute("stroke-dashoffset"))
        : [],
    };
  });

// ── G7 · one base ──────────────────────────────────────────────────────────────────────────
test("the name counts what the strokes count, and the number takes over at six", async ({
  page,
}) => {
  const room = `tally-base-${Date.now()}`;
  await page.goto(`/?size=3&difficulty=EASY&wire=local&s=${room}`);
  await settled(page);

  let at = 1;
  for (const N of [1, 2, 3, 4, 5, 6, 16]) {
    await peers(page, room, N - at);
    at = N;
    const h = await head(page);
    // 2.5.3 BY CONSTRUCTION: the name carries the glyph's digits, because it IS the count.
    expect(h.label).toBe(N === 1 ? "1 player" : `${N} players`);
    expect(h.strokes).toBe(N <= 5 ? N : 0);
    if (N > 5) {
      expect(h.label).toContain(h.written!);
      // THE RUNG IS THE PRICE OF THE SWAP: the numeral sits at `--type-heading` (25.888),
      // not the subheading rung the swap shipped at in pass 2 (20.352).
      expect(parseFloat(h.writtenPx!)).toBeGreaterThan(25);
    }
    // the head line never steps: the mark is its own height at every count
    expect(h.height).toBeGreaterThanOrEqual(36);
  }
});

// ── G3 · the width table ───────────────────────────────────────────────────────────────────
test("the width table, and the floor at the smallest count", async ({ page }) => {
  const room = `tally-width-${Date.now()}`;
  await page.goto(`/?size=3&difficulty=EASY&wire=local&s=${room}`);
  await settled(page);
  const want: Record<number, number> = {
    1: 44,
    2: 44,
    3: 51.66,
    4: 62.28,
    5: 72.92,
    6: 44,
  };
  const got: Record<number, number> = {};
  let at = 1;
  for (const N of [1, 2, 3, 4, 5, 6]) {
    await peers(page, room, N - at);
    at = N;
    const h = await head(page);
    got[N] = h.width!;
    // TO 0.01, not "to within 1.5": pass 2 reported a table to two places and asserted a
    // tolerance that would have passed on a different table.
    expect(Math.abs(h.width! - want[N]), `width at N=${N} (got ${h.width})`).toBeLessThan(0.02);
  }
  console.log("WIDTH TABLE", JSON.stringify(got));
});

// ── G9 · the crossing, both ways and both at once ──────────────────────────────────────────
test("a departure re-draws nothing, and a depart-and-arrive draws exactly the newcomer", async ({
  page,
}) => {
  const room = `tally-cross-${Date.now()}`;
  await page.goto(`/?size=3&difficulty=EASY&wire=local&s=${room}`);
  await settled(page);
  const ids = ["p0", "p1", "p2", "p3", "p4"];
  const say = (msgs: { kind: string; from: string }[]) =>
    page.evaluate(
      ({ room, msgs }) => {
        const w = window as unknown as { __ch?: BroadcastChannel };
        w.__ch ??= new BroadcastChannel(`board:${room}`);
        for (const m of msgs) w.__ch.postMessage({ kind: m.kind, data: {}, from: m.from });
      },
      { room, msgs },
    );

  await say(ids.map((from) => ({ kind: "hi", from })));
  await page.waitForTimeout(900);
  expect((await head(page)).written).toBe("6");

  // SIX TO FIVE: the number gives way to five strokes, and not one of them tweens. A watch
  // that only knows `now > was` re-draws all five here, because `drawn` goes 0 -> 5.
  await say([{ kind: "bye", from: ids[4] }]);
  for (let i = 0; i < 5; i++) {
    await page.waitForTimeout(80);
    const off = (await head(page)).offs;
    expect(off.every((v) => v === "0")).toBe(true);
  }
  const five = await head(page);
  expect(five.strokes).toBe(5);

  // A MIDDLE DEPARTURE: the pose is keyed to the person and the slot is a transform, so four
  // `d` attributes come through byte for byte and only the leaver's is gone.
  await say([{ kind: "bye", from: ids[1] }]);
  await page.waitForTimeout(700);
  const four = await head(page);
  expect(four.strokes).toBe(4);
  expect(four.ds.filter((d) => five.ds.includes(d)).length).toBe(4);

  // ONE TICK, BOTH WAYS — the row pass 2 could not have passed. The count does not move, so a
  // watch on the count never fires and the newcomer inherits the departed person's reveal.
  // Watching the ID SET, exactly one stroke is new and exactly one is gone.
  await say([
    { kind: "bye", from: ids[0] },
    { kind: "hi", from: "p9" },
  ]);
  await page.waitForTimeout(120);
  const mid = await head(page);
  expect(mid.strokes, "the count did not move").toBe(4);
  // exactly one stroke is mid-draw (dashoffset above 0); the other three are untouched
  const drawing = mid.offs.filter((v) => v !== "0").length;
  expect(drawing, `exactly the newcomer draws in (offs ${JSON.stringify(mid.offs)})`).toBe(1);
  await page.waitForTimeout(700);
  const after = await head(page);
  expect(after.ds.filter((d) => four.ds.includes(d)).length).toBe(3);
});

// ── G10/G11 · the seam and the keys ────────────────────────────────────────────────────────
test("Space and Escape, and a mouse press that takes no focus", async ({ page }) => {
  const room = `tally-keys-${Date.now()}`;
  await page.goto(`/?size=3&difficulty=EASY&wire=local&s=${room}`);
  await settled(page);
  await peers(page, room, 2);
  const expanded = () =>
    page.evaluate(
      () => document.querySelector("[data-player-mark]")?.getAttribute("aria-expanded"),
    );

  await page.evaluate(() =>
    (document.querySelector("[data-player-mark]") as HTMLElement).focus(),
  );
  await page.keyboard.press("Space");
  await expect.poll(expanded).toBe("true");
  await page.keyboard.press("Escape");
  await expect.poll(expanded).toBe("false");

  // The caret stays where the reader put it: `@pointerdown.prevent` means a press opens the
  // sheet without taking focus off the board, and Escape is window-bound so it still closes.
  const cell = page.locator(".sudoku-cell input").first();
  await cell.click();
  await mark(page).click();
  await expect.poll(expanded).toBe("true");
  expect(await page.evaluate(() => document.activeElement?.tagName)).toBe("INPUT");
  await page.keyboard.press("Escape");
  await expect.poll(expanded).toBe("false");
  expect(await page.evaluate(() => document.activeElement?.tagName)).toBe("INPUT");
});

// ── G11 · the Tab route, declared per row ──────────────────────────────────────────────────
test("the mark is reachable by Tab", async ({ page, browserName }) => {
  // DECLARED, NOT INHERITED: PW-WebKit does not move focus between controls on a Tab press
  // without the platform's own full-keyboard-access setting, so this row asserts in chromium
  // and says so out loud here rather than reporting a silent green in the other engine.
  test.skip(
    browserName === "webkit",
    "PW-WebKit: Tab does not traverse non-text controls (macOS full keyboard access is off " +
      "in the bundled build); the reachability claim is chromium's, and this row says so.",
  );
  const room = `tally-tab-${Date.now()}`;
  await page.goto(`/?size=3&difficulty=EASY&wire=local&s=${room}`);
  await settled(page);
  await peers(page, room, 2);
  let presses = 0;
  await page.evaluate(() => (document.activeElement as HTMLElement)?.blur());
  for (; presses < 12; presses++) {
    await page.keyboard.press("Tab");
    const onMark = await page.evaluate(
      () => !!document.activeElement?.closest("[data-player-mark]"),
    );
    if (onMark) break;
  }
  console.log(`TAB ROUTE ${browserName}: press ${presses + 1}`);
  expect(presses).toBeLessThan(12);
});

// ── G17 · the solo arm, as shipped ─────────────────────────────────────────────────────────
test("solo: one graphite stroke and a register with a row", async ({ page }) => {
  await page.goto("/?size=3&difficulty=EASY");
  await settled(page);
  const h = await head(page);
  expect(h.label).toBe("1 player");
  expect(h.strokes).toBe(1);

  await mark(page).click();

  await expect(sheet).toBeVisible();

  await expect(sheet.locator(".pl-row")).toHaveCount(1);
  await expect(sheet.locator(".pl-state")).toHaveText("1 player");
});

// ── G8/G7 · the short phone names two people, and has no foot ──────────────────────────────
test.describe("the short phone", () => {
  test.use({ viewport: { width: 390, height: 664 }, hasTouch: true, isMobile: true });
  test("two named rows, no foot", async ({ page }) => {
    const room = `tally-short-${Date.now()}`;
    await page.goto(`/?size=3&difficulty=EASY&wire=local&s=${room}`);
    await settled(page);
    // THE REGIME IS WITNESSED, never assumed.
    const regime = await page.evaluate(() => ({
      coarse: matchMedia("(pointer: coarse)").matches,
      tall: matchMedia("(min-height: 800px)").matches,
    }));
    expect(regime.coarse).toBe(true);
    expect(regime.tall).toBe(false);
    await peers(page, room, 2); // N = 3
    await mark(page).click();
    const sheet = page.locator("[data-lobby]:visible").first();
    await expect(sheet).toBeVisible();
    await settle(page);
    await expect(sheet.locator(".pl-state")).toHaveText("3 players");
    await expect(sheet.locator(".pl-row")).toHaveCount(2);
    await expect(sheet.locator(".pl-more")).toHaveCount(0);
    const named = await sheet.locator(".pl-name").allTextContents();
    console.log("SHORT ROWS", JSON.stringify(named));
    const box = await sheet.boundingBox();
    console.log("SHORT SHEET H", box?.height.toFixed(2));
    // the tap floor, coarse, with the 40px negative control
    const m = await mark(page).boundingBox();
    console.log("COARSE MARK", JSON.stringify(m));
    expect(m!.height).toBeGreaterThanOrEqual(44);
    expect(m!.height).toBeGreaterThan(40);
  });
});

// ── G14 · the sheet's quiet rungs on the opaque ground ─────────────────────────────────────
test.describe("the sheet's ground", () => {
  // 900 tall, so the TALL row budget is the one under test: at the estate's default 720 the
  // register is in the SHORT regime and has no foot at all (see the return's gap on the
  // height-only MQL).
  test.use({ viewport: { width: 1280, height: 900 } });
  test("the register's ground is opaque and its quiet rungs are legible on it", async ({
    page,
  }) => {
  const room = `tally-aa-${Date.now()}`;
  await page.goto(`/?size=3&difficulty=EASY&wire=local&s=${room}`);
  await settled(page);
  await peers(page, room, 15);
  await mark(page).click();
  const read = await page.evaluate(() => {
    const sheet = document.querySelector("[data-lobby]") as HTMLElement;
    const g = (el: Element | null) => (el ? getComputedStyle(el) : null);
    return {
      ground: g(sheet)!.backgroundColor,
      state: g(sheet.querySelector(".pl-state"))!.color,
      more: g(sheet.querySelector(".pl-more"))!.color,
      name: g(sheet.querySelector(".pl-name"))!.color,
    };
  });
  console.log("SHEET COLOURS", JSON.stringify(read));
    // the ground carries no alpha below 1: a list needs a ground it can be measured on
    expect(read.ground).not.toMatch(/,\s*0?\.\d+\)/);
  });
});

// ── PRM · the draw-in snaps ────────────────────────────────────────────────────────────────
test("reduced motion: every stroke lands inked, no tween", async ({ page }) => {
  await page.emulateMedia({ reducedMotion: "reduce" });
  const room = `tally-prm-${Date.now()}`;
  await page.goto(`/?size=3&difficulty=EASY&wire=local&s=${room}`);
  await settled(page);
  await peers(page, room, 3);
  const h = await head(page);
  expect(h.strokes).toBe(4);
  expect(h.offs.every((v) => v === "0")).toBe(true);
});
